MATRIX-MULTIPLY-STRASSEN(A, B, C, n)
    if n == 1
        C11 = C11 + A11 * B11

    partition A into A11, A12, A21, A22
    partition B into B11, B12, B21, B22
    partition C into C11, C12, C21, C22

    create S1..S10 size n/2 x n/2
    create P1..P7 size n/2 x n/2

    S1 = B12 - B22
    S2 = A11 + A12
    S3 = A21 + A22
    S4 = B21 - B11
    S5 = A11 + A22
    S6 = B11 + B22
    S7 = A12 - A22
    S8 = B21 + B22
    S9 = A11 - A21

    MATRIX-MULTIPLY-STRASSEN(A11, S1, P1, n/2)
    MATRIX-MULTIPLY-STRASSEN(S2, B22, P2, n/2)
    MATRIX-MULTIPLY-STRASSEN(S3, B11, P3, n/2)
    MATRIX-MULTIPLY-STRASSEN(A22, S4, P4, n/2)
    MATRIX-MULTIPLY-STRASSEN(S5, S6, P5, n/2)
    MATRIX-MULTIPLY-STRASSEN(S7, S8, P6, n/2)
    MATRIX-MULTIPLY-STRASSEN(S9, S10, P7, n/2)

    C11 = C11 + P5 + P4 - P2 + P6
    C12 = C12 + P1 + P2
    C21 = C21 + P3 + P4
    C22 = C22 + P5 + P1 - P3 - P7
