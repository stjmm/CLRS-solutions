```text
d-MAX-HEAPFIY(A, i)
    largest = i
    for j = 1..d
        if d-CHILD(i, j) < A.heap_size and d-CHILD(i, j) > A[largest]
            largest = d-CHILD(i, j)
    if largest != i
        swap(A[i], A[largest])
        d-MAX-HEAPFIY(A, largest)

d-EXTRACT-MAX(A, d)
    if (A.heap_size < 1)
        error "underflow"

    max = A[1]
    A[1] = A[A.heap_size]
    A.heap_size = A.heap_size - 1
    d-MAX-HEAPFIY(A, 1)
    return max

```

```text
d-INCREASE-MAX(A, i, k)
    if (k < A[i])
        error "new key smaller than curr"

    A[i] = k
    while i > 1 and A[PARENT(i)] < A[i]
        swap A[i], A[PARENT(i)]
        i = PARENT(i)
```
